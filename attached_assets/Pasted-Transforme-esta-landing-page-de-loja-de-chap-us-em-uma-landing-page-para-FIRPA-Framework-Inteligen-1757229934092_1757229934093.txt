// Adicionar no arquivo principal
const calcularMaturidade = () => {
  const processos = document.getElementById('processos').value;
  const tecnologia = document.getElementById('tecnologia').value;
  const equipe = document.getElementById('equipe').value;
  const governanca = document.getElementById('governanca').value;
  
  // Algoritmo baseado na pesquisa
  const score = (processos * 0.3 + tecnologia * 0.25 + 
                 equipe * 0.25 + governanca * 0.2);
  
  // Classificação
  let nivel = '';
  if (score >= 75) nivel = 'Otimizado';
  else if (score >= 50) nivel = 'Avançado';
  else if (score >= 25) nivel = 'Intermediário';
  else nivel = 'Inicial';
  
  // Economia potencial (baseado em dados reais)
  const economia = Math.round(score * 5000); // R$ por mês
  
  // Atualizar interface
  document.getElementById('score').textContent = score + '%';
  document.getElementById('nivel').textContent = nivel;
  document.getElementById('economia').textContent = 'R$ ' + economia.toLocaleString('pt-BR');
  
  // Gerar roadmap dinâmico
  gerarRoadmap(nivel);
};
